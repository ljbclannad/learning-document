## 4. 预热状态管理 (WarmupState.java)

```java
package com.example.warmup.core;

import lombok.Data;
import java.time.Instant;

@Data
public class WarmupState {
    
    public enum Phase {
        NOT_STARTED,
        WARMING_UP,
        COMPLETED
    }
    
    private Phase phase = Phase.NOT_STARTED;
    
    private Instant startTime;
    
    private Instant endTime;
    
    private int currentWeight;
    
    private long requestCount = 0;
    
    private long successCount = 0;
    
    private long failureCount = 0;
    
    public void incrementRequest() {
        requestCount++;
    }
    
    public void incrementSuccess() {
        successCount++;
    }
    
    public void incrementFailure() {
        failureCount++;
    }
    
    public double getSuccessRate() {
        if (requestCount == 0) {
            return 0.0;
        }
        return (double) successCount / requestCount * 100;
    }
    
    public long getElapsedTime() {
        if (startTime == null) {
            return 0;
        }
        Instant now = endTime != null ? endTime : Instant.now();
        return now.toEpochMilli() - startTime.toEpochMilli();
    }
    
    public boolean isWarming() {
        return phase == Phase.WARMING_UP;
    }
    
    public boolean isCompleted() {
        return phase == Phase.COMPLETED;
    }
}
```

## 5. 预热管理器 (WarmupManager.java)

```java
package com.example.warmup.core;

import com.example.warmup.config.WarmupProperties;
import jakarta.annotation.PostConstruct;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;

import java.time.Instant;
import java.util.concurrent.Executors;
import java.util.concurrent.ScheduledExecutorService;
import java.util.concurrent.TimeUnit;

@Slf4j
@Component
@RequiredArgsConstructor
public class WarmupManager {
    
    private final WarmupProperties properties;
    private final WarmupState state = new WarmupState();
    private final ScheduledExecutorService scheduler = Executors.newSingleThreadScheduledExecutor();
    
    @PostConstruct
    public void init() {
        if (properties.isEnabled() && properties.isAutoStart()) {
            log.info("Auto-starting warmup process");
            startWarmup();
        }
    }
    
    public synchronized void startWarmup() {
        if (state.isWarming()) {
            log.warn("Warmup already in progress");
            return;
        }
        
        log.info("Starting warmup process - duration: {}ms, initial weight: {}, target weight: {}",
                properties.getDuration(), properties.getInitialWeight(), properties.getTargetWeight());
        
        state.setPhase(WarmupState.Phase.WARMING_UP);
        state.setStartTime(Instant.now());
        state.setCurrentWeight(properties.getInitialWeight());
        
        scheduleWeightIncrement();
    }
    
    private void scheduleWeightIncrement() {
        scheduler.scheduleAtFixedRate(() -> {
            try {
                incrementWeight();
            } catch (Exception e) {
                log.error("Error during weight increment", e);
            }
        }, properties.getIncrementInterval(), properties.getIncrementInterval(), TimeUnit.MILLISECONDS);
    }
    
    private synchronized void incrementWeight() {
        if (!state.isWarming()) {
            return;
        }
        
        long elapsed = state.getElapsedTime();
        
        if (elapsed >= properties.getDuration()) {
            completeWarmup();
            return;
        }
        
        int newWeight = Math.min(
            state.getCurrentWeight() + properties.getWeightIncrement(),
            properties.getTargetWeight()
        );
        
        state.setCurrentWeight(newWeight);
        
        log.info("Warmup progress - weight: {}/{}, elapsed: {}ms/{}, requests: {}, success rate: {:.2f}%",
                newWeight, properties.getTargetWeight(), elapsed, properties.getDuration(),
                state.getRequestCount(), state.getSuccessRate());
        
        if (newWeight >= properties.getTargetWeight()) {
            completeWarmup();
        }
    }
    
    private synchronized void completeWarmup() {
        if (!state.isWarming()) {
            return;
        }
        
        state.setPhase(WarmupState.Phase.COMPLETED);
        state.setEndTime(Instant.now());
        state.setCurrentWeight(properties.getTargetWeight());
        
        scheduler.shutdown();
        
        log.info("Warmup completed - total time: {}ms, total requests: {}, success rate: {:.2f}%",
                state.getElapsedTime(), state.getRequestCount(), state.getSuccessRate());
    }
    
    public int getCurrentWeight() {
        if (!properties.isEnabled()) {
            return 100;
        }
        
        if (state.isCompleted()) {
            return properties.getTargetWeight();
        }
        
        if (state.isWarming()) {
            return state.getCurrentWeight();
        }
        
        return properties.getInitialWeight();
    }
    
    public WarmupState getState() {
        return state;
    }
    
    public boolean isEnabled() {
        return properties.isEnabled();
    }
    
    public void recordRequest(boolean success) {
        state.incrementRequest();
        if (success) {
            state.incrementSuccess();
        } else {
            state.incrementFailure();
        }
    }
}
```

## 6. 流量控制器 (TrafficController.java)

```java
package com.example.warmup.core;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;

import java.util.Random;

@Slf4j
@Component
@RequiredArgsConstructor
public class TrafficController {
    
    private final WarmupManager warmupManager;
    private final Random random = new Random();
    
    public boolean shouldAcceptRequest() {
        if (!warmupManager.isEnabled()) {
            return true;
        }
        
        int currentWeight = warmupManager.getCurrentWeight();
        
        if (currentWeight >= 100) {
            return true;
        }
        
        int randomValue = random.nextInt(100) + 1;
        boolean accept = randomValue <= currentWeight;
        
        if (!accept) {
            log.debug("Request rejected during warmup - weight: {}, random: {}", currentWeight, randomValue);
        }
        
        return accept;
    }
}
```

## 7. 预热过滤器 (WarmupFilter.java)

```java
package com.example.warmup.filter;

import com.example.warmup.core.TrafficController;
import com.example.warmup.core.WarmupManager;
import jakarta.servlet.*;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.core.annotation.Order;
import org.springframework.stereotype.Component;

import java.io.IOException;

@Slf4j
@Component
@Order(1)
@RequiredArgsConstructor
public class WarmupFilter implements Filter {
    
    private final TrafficController trafficController;
    private final WarmupManager warmupManager;
    
    @Override
    public void doFilter(ServletRequest request, ServletResponse response, FilterChain chain)
            throws IOException, ServletException {
        
        HttpServletRequest httpRequest = (HttpServletRequest) request;
        HttpServletResponse httpResponse = (HttpServletResponse) response;
        
        String path = httpRequest.getRequestURI();
        if (isExcludedPath(path)) {
            chain.doFilter(request, response);
            return;
        }
        
        if (!trafficController.shouldAcceptRequest()) {
            httpResponse.setStatus(HttpServletResponse.SC_SERVICE_UNAVAILABLE);
            httpResponse.setHeader("X-Warmup-Status", "warming-up");
            httpResponse.setHeader("X-Warmup-Weight", String.valueOf(warmupManager.getCurrentWeight()));
            httpResponse.setHeader("Retry-After", "30");
            httpResponse.getWriter().write("{\"error\":\"Service is warming up, please retry later\"}");
            warmupManager.recordRequest(false);
            return;
        }
        
        try {
            chain.doFilter(request, response);
            warmupManager.recordRequest(true);
        } catch (Exception e) {
            warmupManager.recordRequest(false);
            throw e;
        }
    }
    
    private boolean isExcludedPath(String path) {
        return path.startsWith("/actuator") || 
               path.startsWith("/warmup") ||
               path.equals("/health") ||
               path.equals("/info");
    }
}
```

## 9. 预热负载均衡器 (WarmupLoadBalancer.java)

```java
package com.example.warmup.loadbalancer;

import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.ObjectProvider;
import org.springframework.cloud.client.ServiceInstance;
import org.springframework.cloud.client.loadbalancer.DefaultResponse;
import org.springframework.cloud.client.loadbalancer.EmptyResponse;
import org.springframework.cloud.client.loadbalancer.Request;
import org.springframework.cloud.client.loadbalancer.Response;
import org.springframework.cloud.loadbalancer.core.ReactorServiceInstanceLoadBalancer;
import org.springframework.cloud.loadbalancer.core.ServiceInstanceListSupplier;
import reactor.core.publisher.Mono;

import java.time.Instant;
import java.util.List;
import java.util.Random;
import java.util.stream.Collectors;

@Slf4j
public class WarmupLoadBalancer implements ReactorServiceInstanceLoadBalancer {
    
    private final ObjectProvider<ServiceInstanceListSupplier> serviceInstanceListSupplierProvider;
    private final String serviceId;
    private final Random random = new Random();
    
    private static final long WARMUP_DURATION = 300000L;
    
    public WarmupLoadBalancer(ObjectProvider<ServiceInstanceListSupplier> serviceInstanceListSupplierProvider,
                             String serviceId) {
        this.serviceInstanceListSupplierProvider = serviceInstanceListSupplierProvider;
        this.serviceId = serviceId;
    }
    
    @Override
    public Mono<Response<ServiceInstance>> choose(Request request) {
        ServiceInstanceListSupplier supplier = serviceInstanceListSupplierProvider.getIfAvailable();
        if (supplier == null) {
            return Mono.just(new EmptyResponse());
        }
        
        return supplier.get(request).next().map(this::processInstanceResponse);
    }
    
    private Response<ServiceInstance> processInstanceResponse(List<ServiceInstance> instances) {
        if (instances.isEmpty()) {
            log.warn("No instances available for service: {}", serviceId);
            return new EmptyResponse();
        }
        
        List<WeightedInstance> weightedInstances = instances.stream()
                .map(this::calculateWeight)
                .collect(Collectors.toList());
        
        int totalWeight = weightedInstances.stream()
                .mapToInt(WeightedInstance::getWeight)
                .sum();
        
        if (totalWeight == 0) {
            log.warn("Total weight is 0, selecting random instance");
            return new DefaultResponse(instances.get(random.nextInt(instances.size())));
        }
        
        int randomWeight = random.nextInt(totalWeight);
        int currentWeight = 0;
        
        for (WeightedInstance wi : weightedInstances) {
            currentWeight += wi.getWeight();
            if (randomWeight < currentWeight) {
                log.debug("Selected instance: {} with weight: {}", 
                         wi.getInstance().getInstanceId(), wi.getWeight());
                return new DefaultResponse(wi.getInstance());
            }
        }
        
        return new DefaultResponse(weightedInstances.get(0).getInstance());
    }
    
    private WeightedInstance calculateWeight(ServiceInstance instance) {
        String warmupMetadata = instance.getMetadata().get("warmup");
        if (!"true".equals(warmupMetadata)) {
            return new WeightedInstance(instance, 100);
        }
        
        String startTimeStr = instance.getMetadata().get("warmup-start-time");
        String durationStr = instance.getMetadata().get("warmup-duration");
        
        if (startTimeStr == null || "0".equals(startTimeStr)) {
            return new WeightedInstance(instance, 10);
        }
        
        long startTime = Long.parseLong(startTimeStr);
        long duration = durationStr != null ? Long.parseLong(durationStr) : WARMUP_DURATION;
        long elapsed = Instant.now().toEpochMilli() - startTime;
        
        if (elapsed >= duration) {
            return new WeightedInstance(instance, 100);
        }
        
        int weight = (int) (10 + (90 * elapsed / duration));
        log.debug("Instance {} warmup progress: {}ms/{}, weight: {}", 
                 instance.getInstanceId(), elapsed, duration, weight);
        
        return new WeightedInstance(instance, weight);
    }
    
    private static class WeightedInstance {
        private final ServiceInstance instance;
        private final int weight;
        
        public WeightedInstance(ServiceInstance instance, int weight) {
            this.instance = instance;
            this.weight = Math.max(1, weight);
        }
        
        public ServiceInstance getInstance() {
            return instance;
        }
        
        public int getWeight() {
            return weight;
        }
    }
}
```

## 11. 预热监控控制器 (WarmupMonitorController.java)

```java
package com.example.warmup.controller;

import com.example.warmup.core.WarmupManager;
import com.example.warmup.core.WarmupState;
import lombok.RequiredArgsConstructor;
import org.springframework.web.bind.annotation.*;

import java.util.HashMap;
import java.util.Map;

@RestController
@RequestMapping("/warmup")
@RequiredArgsConstructor
public class WarmupMonitorController {
    
    private final WarmupManager warmupManager;
    
    @GetMapping("/status")
    public Map<String, Object> getStatus() {
        WarmupState state = warmupManager.getState();
        
        Map<String, Object> response = new HashMap<>();
        response.put("enabled", warmupManager.isEnabled());
        response.put("phase", state.getPhase());
        response.put("currentWeight", warmupManager.getCurrentWeight());
        response.put("requestCount", state.getRequestCount());
        response.put("successCount", state.getSuccessCount());
        response.put("failureCount", state.getFailureCount());
        response.put("successRate", state.getSuccessRate());
        response.put("elapsedTime", state.getElapsedTime());
        
        if (state.getStartTime() != null) {
            response.put("startTime", state.getStartTime().toString());
        }
        if (state.getEndTime() != null) {
            response.put("endTime", state.getEndTime().toString());
        }
        
        return response;
    }
    
    @PostMapping("/start")
    public Map<String, String> startWarmup() {
        warmupManager.startWarmup();
        
        Map<String, String> response = new HashMap<>();
        response.put("message", "Warmup started");
        return response;
    }
}
```

## 核心机制说明

### 1. 权重计算

- **初始权重**: 10% (只接受 10% 的流量)
- **目标权重**: 100% (接受全部流量)
- **增长策略**: 每 30 秒增加 10%
- **预热时长**: 5 分钟

### 2. 流量控制

使用随机数算法控制流量：
- 生成 1-100 的随机数
- 如果随机数 ≤ 当前权重，则接受请求
- 否则返回 503 状态码

### 3. 负载均衡

客户端负载均衡器会根据服务实例的预热状态分配流量：
- 读取实例的 `warmup-start-time` 元数据
- 计算预热进度
- 按权重分配请求

### 4. 监控指标

- 当前权重
- 请求总数
- 成功/失败次数
- 成功率
- 预热耗时

## 配置调优

### 快速预热（高性能服务）

```yaml
warmup:
  duration: 60000        # 1分钟
  initial-weight: 30     # 初始30%
  weight-increment: 20   # 每次增加20%
  increment-interval: 10000  # 每10秒
```

### 慢速预热（资源密集型服务）

```yaml
warmup:
  duration: 600000       # 10分钟
  initial-weight: 5      # 初始5%
  weight-increment: 5    # 每次增加5%
  increment-interval: 30000  # 每30秒
```

### 禁用预热

```yaml
warmup:
  enabled: false
```

## 注意事项

1. **JVM 预热**: 小流量预热主要解决 JVM 类加载、JIT 编译、连接池初始化等问题
2. **监控告警**: 预热期间的 503 错误是正常的，需要在监控系统中排除
3. **健康检查**: 健康检查接口不应受预热影响，需要在过滤器中排除
4. **元数据同步**: 确保 Nacos 元数据及时更新，避免负载均衡器使用过期数据
5. **客户端重试**: 客户端应实现重试机制，处理预热期间的 503 响应
6. **预热时长**: 根据服务特性调整预热时长，避免过长或过短
7. **权重策略**: 可以根据实际情况调整权重增长策略（线性、指数等）
