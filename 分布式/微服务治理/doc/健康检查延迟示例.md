# 健康检查延迟代码示例

## 概述

在微服务架构中，服务启动后需要一段时间来初始化各种依赖（数据库连接、缓存、消息队列等）。健康检查延迟机制可以避免在服务未完全就绪时就开始接收流量。

---

## 1. Spring Boot 健康检查延迟

### 配置文件 (application.yml)

```yaml
management:
  health:
    livenessState:
      enabled: true
    readinessState:
      enabled: true
  endpoint:
    health:
      probes:
        enabled: true
      show-details: always

# 自定义延迟时间
health:
  check:
    delay: 30000  # 30秒延迟
```

### 自定义健康检查指示器

```java
import org.springframework.beans.factory.annotation.Value;
import org.springframework.boot.actuate.health.Health;
import org.springframework.boot.actuate.health.HealthIndicator;
import org.springframework.stereotype.Component;

@Component
public class DelayedHealthIndicator implements HealthIndicator {
    
    private final long startTime;
    private final long delayMillis;
    
    public DelayedHealthIndicator(@Value("${health.check.delay:30000}") long delayMillis) {
        this.startTime = System.currentTimeMillis();
        this.delayMillis = delayMillis;
    }
    
    @Override
    public Health health() {
        long uptime = System.currentTimeMillis() - startTime;
        
        if (uptime < delayMillis) {
            return Health.down()
                    .withDetail("status", "warming up")
                    .withDetail("uptime", uptime)
                    .withDetail("required", delayMillis)
                    .build();
        }
        
        return Health.up()
                .withDetail("uptime", uptime)
                .build();
    }
}
```

## 2. Kubernetes Probe 延迟配置

```yaml
apiVersion: v1
kind: Pod
metadata:
  name: my-service
spec:
  containers:
  - name: app
    image: my-app:latest
    
    # 存活探针 - 检测容器是否需要重启
    livenessProbe:
      httpGet:
        path: /actuator/health/liveness
        port: 8080
      initialDelaySeconds: 30    # 启动后延迟30秒开始检查
      periodSeconds: 10           # 每10秒检查一次
      timeoutSeconds: 5           # 超时时间5秒
      failureThreshold: 3         # 连续失败3次判定为不健康
      
    # 就绪探针 - 检测容器是否可以接收流量
    readinessProbe:
      httpGet:
        path: /actuator/health/readiness
        port: 8080
      initialDelaySeconds: 20    # 启动后延迟20秒开始检查
      periodSeconds: 5            # 每5秒检查一次
      timeoutSeconds: 3           # 超时时间3秒
      failureThreshold: 3         # 连续失败3次判定为未就绪
      successThreshold: 1         # 成功1次即判定为就绪
      
    # 启动探针 - 检测容器是否已启动（适用于启动慢的应用）
    startupProbe:
      httpGet:
        path: /actuator/health/liveness
        port: 8080
      initialDelaySeconds: 0
      periodSeconds: 10
      timeoutSeconds: 3
      failureThreshold: 30        # 最多等待300秒（30次 * 10秒）
```

### Probe 类型说明

- **livenessProbe（存活探针）**：检测容器是否在运行，失败会重启容器
- **readinessProbe（就绪探针）**：检测容器是否准备好接收流量，失败会从 Service 中移除
- **startupProbe（启动探针）**：检测容器是否已启动，在启动探针成功前，其他探针不会执行

## 3. 自定义健康检查服务

### 健康检查服务

```java
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Service;

import javax.annotation.PostConstruct;
import java.util.ArrayList;
import java.util.List;
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.CopyOnWriteArrayList;
import java.util.concurrent.atomic.AtomicBoolean;

@Service
public class HealthCheckService {
    
    private static final Logger log = LoggerFactory.getLogger(HealthCheckService.class);
    
    private final AtomicBoolean ready = new AtomicBoolean(false);
    private final List<String> dependencies = new CopyOnWriteArrayList<>();
    
    @PostConstruct
    public void init() {
        // 异步执行延迟健康检查
        CompletableFuture.runAsync(() -> {
            try {
                // 等待服务预热
                Thread.sleep(5000);
                log.info("Service warming up...");
                
                // 检查数据库连接
                checkDatabase();
                dependencies.add("database");
                log.info("Database connection ready");
                
                // 检查缓存连接
                checkCache();
                dependencies.add("cache");
                log.info("Cache connection ready");
                
                // 检查消息队列
                checkMessageQueue();
                dependencies.add("mq");
                log.info("Message queue connection ready");
                
                // 所有依赖就绪
                ready.set(true);
                log.info("Health check completed, service is ready");
                
            } catch (Exception e) {
                log.error("Health check failed", e);
            }
        });
    }
    
    public boolean isReady() {
        return ready.get();
    }
    
    public List<String> getReadyDependencies() {
        return new ArrayList<>(dependencies);
    }
    
    private void checkDatabase() throws Exception {
        // 数据库连接检查逻辑
        Thread.sleep(2000);
        // 实际实现：执行简单查询如 SELECT 1
    }
    
    private void checkCache() throws Exception {
        // 缓存连接检查逻辑
        Thread.sleep(1000);
        // 实际实现：执行 PING 命令
    }
    
    private void checkMessageQueue() throws Exception {
        // 消息队列检查逻辑
        Thread.sleep(1000);
        // 实际实现：检查连接状态
    }
}
```

### 健康检查控制器

```java
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

import java.util.HashMap;
import java.util.Map;

@RestController
@RequestMapping("/health")
public class HealthController {
    
    @Autowired
    private HealthCheckService healthCheckService;
    
    /**
     * 就绪检查 - 用于判断是否可以接收流量
     */
    @GetMapping("/ready")
    public ResponseEntity<Map<String, Object>> readiness() {
        Map<String, Object> response = new HashMap<>();
        
        if (healthCheckService.isReady()) {
            response.put("status", "UP");
            response.put("dependencies", healthCheckService.getReadyDependencies());
            return ResponseEntity.ok(response);
        } else {
            response.put("status", "DOWN");
            response.put("message", "Service is warming up");
            return ResponseEntity.status(503).body(response);
        }
    }
    
    /**
     * 存活检查 - 用于判断进程是否需要重启
     */
    @GetMapping("/live")
    public ResponseEntity<String> liveness() {
        // 存活检查通常更简单，只要进程在运行就返回成功
        return ResponseEntity.ok("OK");
    }
}
```

## 4. 渐进式健康检查

适用于需要分阶段初始化的复杂服务。

```java
import org.springframework.stereotype.Component;

import javax.annotation.PostConstruct;
import java.util.concurrent.Executors;
import java.util.concurrent.ScheduledExecutorService;
import java.util.concurrent.TimeUnit;

@Component
public class GradualHealthCheck {
    
    /**
     * 健康检查阶段
     */
    private enum HealthStage {
        STARTING(0, "服务启动中"),
        DATABASE_READY(30, "数据库已就绪"),
        CACHE_READY(50, "缓存已就绪"),
        MQ_READY(70, "消息队列已就绪"),
        FULLY_READY(100, "完全就绪");
        
        private final int percentage;
        private final String description;
        
        HealthStage(int percentage, String description) {
            this.percentage = percentage;
            this.description = description;
        }
        
        public int getPercentage() {
            return percentage;
        }
        
        public String getDescription() {
            return description;
        }
    }
    
    private volatile HealthStage currentStage = HealthStage.STARTING;
    
    @PostConstruct
    public void startHealthCheck() {
        ScheduledExecutorService executor = Executors.newSingleThreadScheduledExecutor();
        
        // 5秒后检查数据库
        executor.schedule(() -> {
            if (checkDatabase()) {
                currentStage = HealthStage.DATABASE_READY;
                System.out.println("Stage: " + currentStage.getDescription());
            }
        }, 5, TimeUnit.SECONDS);
        
        // 10秒后检查缓存
        executor.schedule(() -> {
            if (checkCache()) {
                currentStage = HealthStage.CACHE_READY;
                System.out.println("Stage: " + currentStage.getDescription());
            }
        }, 10, TimeUnit.SECONDS);
        
        // 15秒后检查消息队列
        executor.schedule(() -> {
            if (checkMessageQueue()) {
                currentStage = HealthStage.MQ_READY;
                System.out.println("Stage: " + currentStage.getDescription());
            }
        }, 15, TimeUnit.SECONDS);
        
        // 20秒后完全就绪
        executor.schedule(() -> {
            currentStage = HealthStage.FULLY_READY;
            System.out.println("Stage: " + currentStage.getDescription());
            executor.shutdown();
        }, 20, TimeUnit.SECONDS);
    }
    
    public int getHealthPercentage() {
        return currentStage.getPercentage();
    }
    
    public String getHealthDescription() {
        return currentStage.getDescription();
    }
    
    public boolean isFullyReady() {
        return currentStage == HealthStage.FULLY_READY;
    }
    
    private boolean checkDatabase() {
        // 数据库检查逻辑
        return true;
    }
    
    private boolean checkCache() {

        return true;
    }
    
    private boolean checkMessageQueue() {

    }
}
```

## 5. 完整的健康检查端点示例

```java
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

import java.time.Duration;
import java.time.Instant;
import java.util.HashMap;
import java.util.Map;

@RestController
@RequestMapping("/actuator/health")
public class DetailedHealthController {
    
    private final Instant startTime = Instant.now();
    
    @Autowired
    private GradualHealthCheck gradualHealthCheck;
    
    @Autowired
    private HealthCheckService healthCheckService;
    
    /**
     * 详细健康检查端点
     */
    @GetMapping
    public ResponseEntity<Map<String, Object>> health() {
        Map<String, Object> health = new HashMap<>();
        
        // 基本信息
        Duration uptime = Duration.between(startTime, Instant.now());
        health.put("status", gradualHealthCheck.isFullyReady() ? "UP" : "WARMING_UP");
        health.put("uptime", uptime.getSeconds() + "s");
        
        // 渐进式健康检查信息
        Map<String, Object> gradual = new HashMap<>();
        gradual.put("percentage", gradualHealthCheck.getHealthPercentage());
        gradual.put("stage", gradualHealthCheck.getHealthDescription());
        health.put("gradual", gradual);
        
        // 依赖检查信息
        Map<String, Object> dependencies = new HashMap<>();
        dependencies.put("ready", healthCheckService.isReady());
        dependencies.put("components", healthCheckService.getReadyDependencies());
        health.put("dependencies", dependencies);
        
        // 根据就绪状态返回不同的 HTTP 状态码
        HttpStatus status = gradualHealthCheck.isFullyReady() 
            ? HttpStatus.OK 
            : HttpStatus.SERVICE_UNAVAILABLE;
            
        return ResponseEntity.status(status).body(health);
    }
    
    /**
     * 简单存活检查
     */
    @GetMapping("/liveness")
    public ResponseEntity<Map<String, String>> liveness() {
        Map<String, String> response = new HashMap<>();
        response.put("status", "UP");
        return ResponseEntity.ok(response);
    }
    
    /**
     * 就绪检查
     */
    @GetMapping("/readiness")
    public ResponseEntity<Map<String, Object>> readiness() {
        Map<String, Object> response = new HashMap<>();
        
        boolean ready = gradualHealthCheck.isFullyReady() && healthCheckService.isReady();
        response.put("status", ready ? "UP" : "DOWN");
        response.put("percentage", gradualHealthCheck.getHealthPercentage());
        
        HttpStatus status = ready ? HttpStatus.OK : HttpStatus.SERVICE_UNAVAILABLE;
        return ResponseEntity.status(status).body(response);
    }
}
```

## 6. 最佳实践

### 延迟时间设置建议

| 检查类型 | 建议延迟时间 | 说明 |
|---------|------------|------|
| startupProbe | 0-10s | 启动探针应尽早开始，但给予足够的失败次数 |
| livenessProbe | 30-60s | 给予应用充分的启动时间 |
| readinessProbe | 20-40s | 比存活探针稍早，确保流量只路由到就绪实例 |

### 超时和阈值设置

```yaml
# 推荐配置
livenessProbe:
  initialDelaySeconds: 30
  periodSeconds: 10
  timeoutSeconds: 5
  failureThreshold: 3      # 30秒内连续失败3次才重启
  successThreshold: 1      # 成功1次即恢复

readinessProbe:
  initialDelaySeconds: 20
  periodSeconds: 5
  timeoutSeconds: 3
  failureThreshold: 3      # 15秒内连续失败3次才标记为未就绪
  successThreshold: 1      # 成功1次即标记为就绪
```

### 注意事项

1. **避免过短的延迟**：可能导致服务未完全初始化就开始接收流量
2. **避免过长的延迟**：会延长服务发布时间，影响部署效率
3. **区分存活和就绪**：存活检查失败会重启容器，就绪检查失败只是暂时不接收流量
4. **监控健康检查**：记录健康检查的响应时间和失败率，及时发现问题
5. **依赖检查要快**：健康检查端点应该快速响应，避免复杂的业务逻辑
