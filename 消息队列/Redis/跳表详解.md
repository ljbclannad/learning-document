# 1. zset（Sorted Set）概述
- 元素唯一，类似 set；
- 每个元素关联一个 **score（double）**，按 score 自动排序；
- 常见命令：ZADD、ZRANGE、ZREM、ZINCRBY 等。

# 2. 底层数据结构设计
Redis 为了兼顾性能和空间，在不同规模下使用两种底层结构：
| 编码方式                                        | 使用条件         | 底层结构           |
| ----------------------------------------------- | ---------------- | ------------------ |
| ziplist（< Redis 5.0）或 listpack（Redis 5.0+） | 元素少、短字符串 | 压缩表（空间紧凑） |
| skiplist                                        | 元素多，性能优先 | 哈希表 + 跳表      |

# 3. zset 结构体定义（跳表编码）
``` c
typedef struct zset {
    dict *dict;         // 快速查找 value -> score（哈希表）提供 O(1) 查找功能
    zskiplist *zsl;     // score 排序结构（跳表） zskiplist：按 score 排序，支持快速范围查询（如 ZRANGE、ZRANK）；
} zset;

typedef struct zskiplist {
    struct zskiplistNode *header, *tail;
    unsigned long length;
    int level;
} zskiplist;

typedef struct zskiplistNode {
    sds ele;                 // 元素内容
    double score;            // 分值
    struct zskiplistNode *backward; /* level0 的反向指针，支持逆序遍历 */
    struct zskiplistLevel {
        struct zskiplistNode *forward; /* 当前层的 forward 指针 */
        unsigned int span; /* 当前层的跨度 */
    } level[]; /* 当前节点的 level 数组，每个 level 包含 forward 和 span */
} zskiplistNode;
```

# 4. 关键操作源码解析

### 4.1 ZADD：添加元素
``` c
/* 插入 (score, ele)，返回新节点指针 */
zskiplistNode *zslInsert(zskiplist *zsl, double score, const char *ele) {
    zskiplistNode *update[ZSKIPLIST_MAXLEVEL];
    unsigned long rank[ZSKIPLIST_MAXLEVEL];

    zskiplistNode *x = zsl->header;

    /* 1) 从最高层往下走：找每一层“最后一个 < (score,ele) 的节点” */
    for (int i = zsl->level - 1; i >= 0; i--) {
        /* rank[i]：到达 update[i] 时，已经跨过的 level0 节点数 */
        rank[i] = (i == zsl->level - 1) ? 0 : rank[i + 1];

        while (x->level[i].forward &&
               zslNodeCompare(x->level[i].forward->score, x->level[i].forward->ele,
                              score, ele) < 0)
        {
            rank[i] += x->level[i].span; /* span 累加：跨过多少个 level0 节点 */
            x = x->level[i].forward;
        }
        update[i] = x;
    }

    /* 2) 随机层高 */
    int level = zslRandomLevel();

    /* 如果新节点层数超过当前最高层：补齐 update/rank，并初始化 header 的 span */
    if (level > zsl->level) {
        for (int i = zsl->level; i < level; i++) {
            rank[i] = 0;
            update[i] = zsl->header;
            /* 这层原本没有索引，新节点会成为第一个 forward，所以 span=length */
            update[i]->level[i].span = zsl->length;
        }
        zsl->level = level;
    }

    /* 3) 创建节点并“逐层插入” */
    x = zslCreateNode(level, score, ele);

    for (int i = 0; i < level; i++) {
        /* 经典链表插入：x 接到 update[i] 后面 */
        x->level[i].forward = update[i]->level[i].forward;
        update[i]->level[i].forward = x;

        /*
         * span 的精髓：
         * - update[i]->span 表示 update[i] 到它原 forward 跨过多少 level0 节点
         * - 插入 x 以后，这个跨度要被“切开”
         *
         * rank[0]：在 level0 上，新节点左侧跨过的节点数
         * rank[i]：在第 i 层上，update[i] 左侧跨过的节点数
         *
         * (rank[0] - rank[i]) 就是：从 update[i] 到 x 的 level0 距离
         */
        x->level[i].span = update[i]->level[i].span - (rank[0] - rank[i]);
        update[i]->level[i].span = (rank[0] - rank[i]) + 1;
    }

    /* 4) 对于没被新节点覆盖到的更高层：它们的 span 也要 +1（因为 level0 多了一个节点） */
    for (int i = level; i < zsl->level; i++) {
        update[i]->level[i].span++;
    }

    /* 5) 维护 backward / tail（只在 level0 维护 backward） */
    x->backward = (update[0] == zsl->header) ? NULL : update[0];

    if (x->level[0].forward) {
        x->level[0].forward->backward = x;
    } else {
        zsl->tail = x;
    }

    zsl->length++;
    return x;
}
```

> 随机跳表的高度分布类似于平衡树，但实现更简单高效。

### 4.2 ZRANGE：按 score 排序范围查询

```c
static int zslValueGteMin(double value, const zrangespec *range) {
    return range->minex ? (value > range->min) : (value >= range->min);
}
static int zslValueLteMax(double value, const zrangespec *range) {
    return range->maxex ? (value < range->max) : (value <= range->max);
}

static int zslIsInRange(const zskiplist *zsl, const zrangespec *range) {
    if (range->min > range->max) return 0;
    if (range->min == range->max && (range->minex || range->maxex)) return 0;

    if (zsl->tail == NULL) return 0;

    /* 最小值都比尾节点还大 => 不可能有 */
    if (!zslValueGteMin(zsl->tail->score, range)) return 0;

    /* 最大值都比首节点还小 => 不可能有（首节点是 header->level0->forward） */
    zskiplistNode *first = zsl->header->level[0].forward;
    if (!first || !zslValueLteMax(first->score, range)) return 0;

    return 1;
}

/* 找到范围内第一个节点（按 score） */
zskiplistNode *zslFirstInRange(zskiplist *zsl, const zrangespec *range) {
    if (!zslIsInRange(zsl, range)) return NULL;

    zskiplistNode *x = zsl->header;

    for (int i = zsl->level - 1; i >= 0; i--) {
        while (x->level[i].forward &&
               !zslValueGteMin(x->level[i].forward->score, range))
        {
            x = x->level[i].forward;
        }
    }

    x = x->level[0].forward;
    if (!x) return NULL;
    if (!zslValueLteMax(x->score, range)) return NULL;
    return x;
}

/* 示例：打印 score 范围内的元素 */
void zslRangeByScorePrint(zskiplist *zsl, double min, double max, int minex, int maxex) {
    zrangespec range = {.min=min, .max=max, .minex=minex, .maxex=maxex};
    zskiplistNode *x = zslFirstInRange(zsl, &range);
    while (x && zslValueLteMax(x->score, &range)) {
        printf("(%f, %s)\n", x->score, x->ele);
        x = x->level[0].forward; /* 只需要 level0 顺序扫 */
    }
}
```

### 4.3 ZRANGE：按rank排序范围查询

```c
/*
 * 取第 rank 个元素（rank 从 1 开始）
 * Redis：ZRANGE 的 start/stop 会先换算成 rank 再调用类似逻辑。
 */
zskiplistNode *zslGetElementByRank(zskiplist *zsl, unsigned long rank) {
    if (rank == 0 || rank > zsl->length) return NULL;

    unsigned long traversed = 0;
    zskiplistNode *x = zsl->header;

    for (int i = zsl->level - 1; i >= 0; i--) {
        while (x->level[i].forward &&
               (traversed + x->level[i].span) <= rank)
        {
            traversed += x->level[i].span;
            x = x->level[i].forward;
        }
        if (traversed == rank) {
            return x;
        }
    }
    return NULL;
}

/* 示例：打印 rank 范围 [start, stop]（start/stop 从 0 开始更像 ZRANGE） */
void zslRangeByRankPrint(zskiplist *zsl, long start, long stop) {
    if (start < 0) start = 0;
    if (stop < 0) return;
    if ((unsigned long)start >= zsl->length) return;
    if ((unsigned long)stop >= zsl->length) stop = (long)zsl->length - 1;

    unsigned long rank = (unsigned long)start + 1; /* 转成 1-based rank */
    zskiplistNode *x = zslGetElementByRank(zsl, rank);

    for (long i = start; x && i <= stop; i++) {
        printf("#%ld -> (%f, %s)\n", i, x->score, x->ele);
        x = x->level[0].forward;
    }
}
```

### 4.4 ZREM：删除元素

```c
/* 内部删除：update[] 已经是每层的前驱，x 是要删的节点 */
static void zslDeleteNode(zskiplist *zsl, zskiplistNode *x, zskiplistNode **update) {
    for (int i = 0; i < zsl->level; i++) {
        if (update[i]->level[i].forward == x) {
            /* 这一层确实指向 x：断开并把 span“拼回去” */
            update[i]->level[i].span += x->level[i].span - 1;
            update[i]->level[i].forward = x->level[i].forward;
        } else {
            /*
             * 这一层没直接指向 x，但 level0 少了一个节点，
             * 所以这层路径上的 span 也要 -1
             */
            update[i]->level[i].span -= 1;
        }
    }

    /* 修 backward / tail */
    if (x->level[0].forward) {
        x->level[0].forward->backward = x->backward;
    } else {
        zsl->tail = x->backward;
    }

    /* 如果最高层已经空了，降低 zsl->level（保持 level 紧凑） */
    while (zsl->level > 1 && zsl->header->level[zsl->level - 1].forward == NULL) {
        zsl->level--;
    }

    zsl->length--;
}

/* 删除精确元素 (score, ele)，成功返回 1，否则 0 */
int zslDelete(zskiplist *zsl, double score, const char *ele) {
    zskiplistNode *update[ZSKIPLIST_MAXLEVEL];
    zskiplistNode *x = zsl->header;

    /* 1) 找到每一层的前驱 */
    for (int i = zsl->level - 1; i >= 0; i--) {
        while (x->level[i].forward &&
               zslNodeCompare(x->level[i].forward->score, x->level[i].forward->ele,
                              score, ele) < 0)
        {
            x = x->level[i].forward;
        }
        update[i] = x;
    }

    /* 2) 候选节点在 level0 的 forward */
    x = x->level[0].forward;

    /* 3) 必须精确匹配 (score, ele) */
    if (x && score == x->score && strcmp(x->ele, ele) == 0) {
        zslDeleteNode(zsl, x, update);
        free(x->ele);
        free(x);
        return 1;
    }
    return 0;
}
```

# 5. Java 版本实现

### 5.1 核心数据结构

```java
public class SkipList<T> {
    private static final int MAX_LEVEL = 32;
    private static final double P = 0.25;
    
    private final Node<T> header;
    private Node<T> tail;
    private int level;
    private long length;
    private final Random random;
    
    static class Node<T> {
        final double score;
        final T value;
        Node<T> backward;
        final Level[] levels;
        
        static class Level {
            Node<?> forward;
            long span;
            
            Level() {
                this.forward = null;
                this.span = 0;
            }
        }
        
        Node(int level, double score, T value) {
            this.score = score;
            this.value = value;
            this.levels = new Level[level];
            for (int i = 0; i < level; i++) {
                this.levels[i] = new Level();
            }
        }
    }
    
    public SkipList() {
        this.header = new Node<>(MAX_LEVEL, 0, null);
        this.tail = null;
        this.level = 1;
        this.length = 0;
        this.random = new Random();
    }
    
    private int randomLevel() {
        int level = 1;
        while (random.nextDouble() < P && level < MAX_LEVEL) {
            level++;
        }
        return level;
    }
    
    private int compareNode(double score1, T value1, double score2, T value2) {
        if (score1 != score2) {
            return Double.compare(score1, score2);
        }
        if (value1 == null || value2 == null) {
            return 0;
        }
        return value1.toString().compareTo(value2.toString());
    }
}
```

### 5.2 插入操作

```java
public void insert(double score, T value) {
    Node<T>[] update = new Node[MAX_LEVEL];
    long[] rank = new long[MAX_LEVEL];
    
    Node<T> x = header;
    
    // 从最高层往下找每一层的前驱节点
    for (int i = level - 1; i >= 0; i--) {
        rank[i] = (i == level - 1) ? 0 : rank[i + 1];
        
        while (x.levels[i].forward != null &&
               compareNode(x.levels[i].forward.score, 
                          (T) x.levels[i].forward.value,
                          score, value) < 0) {
            rank[i] += x.levels[i].span;
            x = (Node<T>) x.levels[i].forward;
        }
        update[i] = x;
    }
    
    // 随机生成层高
    int newLevel = randomLevel();
    
    // 如果新层高超过当前最大层高，初始化新层
    if (newLevel > level) {
        for (int i = level; i < newLevel; i++) {
            rank[i] = 0;
            update[i] = header;
            update[i].levels[i].span = length;
        }
        level = newLevel;
    }
    
    // 创建新节点
    x = new Node<>(newLevel, score, value);
    
    // 逐层插入
    for (int i = 0; i < newLevel; i++) {
        x.levels[i].forward = update[i].levels[i].forward;
        update[i].levels[i].forward = x;
        
        // 更新 span
        x.levels[i].span = update[i].levels[i].span - (rank[0] - rank[i]);
        update[i].levels[i].span = (rank[0] - rank[i]) + 1;
    }
    
    // 更新未被新节点覆盖的层的 span
    for (int i = newLevel; i < level; i++) {
        update[i].levels[i].span++;
    }
    
    // 维护 backward 指针
    x.backward = (update[0] == header) ? null : update[0];
    if (x.levels[0].forward != null) {
        ((Node<T>) x.levels[0].forward).backward = x;
    } else {
        tail = x;
    }
    
    length++;
}
```

### 5.3 按 Score 范围查询

```java
public List<T> rangeByScore(double min, double max, 
                            boolean minExclusive, boolean maxExclusive) {
    List<T> result = new ArrayList<>();
    
    if (!isInRange(min, max, minExclusive, maxExclusive)) {
        return result;
    }
    
    Node<T> x = firstInRange(min, max, minExclusive, maxExclusive);
    
    while (x != null && valueLteMax(x.score, max, maxExclusive)) {
        result.add(x.value);
        x = (Node<T>) x.levels[0].forward;
    }
    
    return result;
}

private boolean isInRange(double min, double max, 
                         boolean minExclusive, boolean maxExclusive) {
    if (min > max || (min == max && (minExclusive || maxExclusive))) {
        return false;
    }
    
    if (tail == null) {
        return false;
    }
    
    if (!valueGteMin(tail.score, min, minExclusive)) {
        return false;
    }
    
    Node<T> first = (Node<T>) header.levels[0].forward;
    if (first == null || !valueLteMax(first.score, max, maxExclusive)) {
        return false;
    }
    
    return true;
}

private Node<T> firstInRange(double min, double max, 
                             boolean minExclusive, boolean maxExclusive) {
    if (!isInRange(min, max, minExclusive, maxExclusive)) {
        return null;
    }
    
    Node<T> x = header;
    
    for (int i = level - 1; i >= 0; i--) {
        while (x.levels[i].forward != null &&
               !valueGteMin(x.levels[i].forward.score, min, minExclusive)) {
            x = (Node<T>) x.levels[i].forward;
        }
    }
    
    x = (Node<T>) x.levels[0].forward;
    
    if (x == null || !valueLteMax(x.score, max, maxExclusive)) {
        return null;
    }
    
    return x;
}

private boolean valueGteMin(double value, double min, boolean exclusive) {
    return exclusive ? (value > min) : (value >= min);
}

private boolean valueLteMax(double value, double max, boolean exclusive) {
    return exclusive ? (value < max) : (value <= max);
}
```

public List<T> rangeByRank(long start, long stop) {
    List<T> result = new ArrayList<>();
    
    if (start < 0) start = 0;
    if (stop < 0) return result;
    if (start >= length) return result;
    if (stop >= length) stop = length - 1;
    
    long rank = start + 1;
    Node<T> x = getNodeByRank(rank);

        x = (Node<T>) x.levels[0].forward;

    return result;
}

### 5.5 删除操作

```java
public boolean delete(double score, T value) {
    Node<T>[] update = new Node[MAX_LEVEL];
    Node<T> x = header;
    
    // 找到每一层的前驱节点
    for (int i = level - 1; i >= 0; i--) {
        while (x.levels[i].forward != null &&
               compareNode(x.levels[i].forward.score,
                          (T) x.levels[i].forward.value,
                          score, value) < 0) {
            x = (Node<T>) x.levels[i].forward;
        }
        update[i] = x;
    }
    
    // 候选节点
    x = (Node<T>) x.levels[0].forward;
    
    // 精确匹配
    if (x != null && x.score == score && 
        (value == null || value.equals(x.value))) {
        deleteNode(x, update);
        return true;
    }
    
    return false;
}

private void deleteNode(Node<T> x, Node<T>[] update) {
    // 更新每一层的指针和 span
    for (int i = 0; i < level; i++) {
        if (update[i].levels[i].forward == x) {
            update[i].levels[i].span += x.levels[i].span - 1;
            update[i].levels[i].forward = x.levels[i].forward;
        } else {
            update[i].levels[i].span--;
        }
    }
    
    // 更新 backward 指针
    if (x.levels[0].forward != null) {
        ((Node<T>) x.levels[0].forward).backward = x.backward;
    } else {
        tail = x.backward;
    }
    
    // 降低层高
    while (level > 1 && header.levels[level - 1].forward == null) {
        level--;
    }
    
    length--;
}

public long size() {
    return length;
}

public int getLevel() {
    return level;
}
```

public class SkipListDemo {
    public static void main(String[] args) {
        SkipList<String> skipList = new SkipList<>();
        
        // 插入元素
        skipList.insert(1.0, "apple");
        skipList.insert(2.5, "banana");
        skipList.insert(3.0, "cherry");
        skipList.insert(1.5, "date");
        skipList.insert(4.0, "elderberry");
        
        System.out.println("Size: " + skipList.size());
        System.out.println("Level: " + skipList.getLevel());
        
        // 按 score 范围查询
        System.out.println("\nRange by score [1.5, 3.0]:");
        List<String> range1 = skipList.rangeByScore(1.5, 3.0, false, false);
        range1.forEach(System.out::println);
        
        // 按 rank 查询
        System.out.println("\nRange by rank [0, 2]:");
        List<String> range2 = skipList.rangeByRank(0, 2);
        range2.forEach(System.out::println);
        
        // 获取指定 rank 的元素
        System.out.println("\nElement at rank 3: " + skipList.getByRank(3));
        
        // 删除元素
        boolean deleted = skipList.delete(2.5, "banana");
        System.out.println("\nDeleted banana: " + deleted);
        System.out.println("Size after deletion: " + skipList.size());
    }
}
